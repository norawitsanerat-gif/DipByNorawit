def imread(filename, flags=cv2.IMREAD_COLOR):
    img = cv2.imread(filename, flags)
    if img is not None: return img
    try:
        return cv2.imdecode(
            np.fromfile(filename, dtype=np.uint8), flags)
    except Exception as e:
        print("error:", e); return None

LAB 3 · lab3_2

Contrast Stretch + Histogram Equalization

ยืดช่วงค่า (piecewise linear) และกระจาย histogram ให้เต็ม 0–255

def pixelVal(pix, r1, s1, r2, s2):
    if 0 <= pix <= r1:
        return (s1 / r1) * pix
    elif r1 < pix <= r2:
        return ((s2 - s1) / (r2 - r1)) * (pix - r1) + s1
    else:
        return ((255 - s2) / (255 - r2)) * (pix - r2) + s2

def contrast_stretch(img, r1, s1, r2, s2):
    return np.vectorize(
        lambda p: pixelVal(p, r1, s1, r2, s2)
    )(img).astype(np.uint8)

result = contrast_stretch(gray, 70, 20, 180, 230)

equalized = cv2.equalizeHist(gray)   # histogram equalization

np.vectorize

เอา function ไปรันทุกพิกเซล — สุดท้าย .astype(np.uint8) เสมอ

กับดัก

equalizeHist ใช้ได้กับ grayscale เท่านั้น — ภาพสีต้อง cvtColor เป็น YUV แล้วทำช่อง Y

เทคนิคสอบโจทย์ "ยืดช่วง 50–150 → 0–255" = contrast stretch ด้วย r1=50, r2=150, s1=0, s2=255

WEEK 4 · Ex1–Ex2

Uniform Quantization + Downsampling

ลดระดับเทา (grey levels) และลดความละเอียดเชิงพื้นที่

def quantize(img, levels, option=1):
    step = 256 // levels
    if option == 0:                      # Min
        q = (img // step) * step
    elif option == 1:                    # Mid (ค่ากลางช่วง)
        q = (img // step) * step + step // 2
    elif option == 2:                    # Max
        q = np.minimum((img // step) * step + step - 1, 255)
    return q.astype(np.uint8)

# 16 ระดับ: q = (img // 16) * 16 + 7

# Downsampling — เลือกทุก ๆ n พิกเซล
z = cv2.resize(cv2.imread("cameraman.tif", 0), (1024, 1024))
a = z[::2, ::2]      # ทุก 2 → ¼ พิกเซล
b = z[::4, ::4]      # ทุก 4 → 1/16 พิกเซล
e = z[::16, ::16];  d = z[::32, ::32]

Min / Mid / Max

ตัวแทนช่วง: Min = ต้นช่วง · Mid = กลางช่วง (step//2) · Max = ปลายช่วง

Ex1 สูตรที่ต้องจำ

q = (img // 16) * 16 + 7 → 210→215 · 108→103 (Mid ของ 16 ระดับ)

เทคนิคสอบลดครึ่ง 2 ด้าน = พิกเซลเหลือ ¼ (กับดัก: ไม่ใช่ ½)

WEEK 4 · Ex quiz

Watermark — ฝังลายน้ำใน Bit-Plane

สร้าง layer ข้อความ → ปิดบิตเป้าหมายด้วย mask → OR ฝัง watermark ลงไป

from PIL import Image, ImageDraw, ImageFont

def create_watermark(text, shape, position=(50,50), alpha=255):
    h, w = shape
    wm = Image.new("L", (w, h), 0)
    draw = ImageDraw.Draw(wm)
    try: font = ImageFont.truetype("arial.ttf", 30)
    except: font = ImageFont.load_default()
    draw.text(position, text, fill=alpha, font=font)
    return np.array(wm)

wm = create_watermark("Norawit S.", img.shape, (50, 100), 255)
wm_bin = (wm > 128).astype(np.uint8)

embed_bit = 7
mask = np.uint8(255 ^ (1 << embed_bit))   # เปิดทุกบิต ยกเว้นบิตเป้า
result = img.copy()
result = (result & mask) | (wm_bin << embed_bit)

# ดึงกลับ/ตรวจ
bit_plane = ((result >> embed_bit) & 1) * 255

หลักการ

255 ^ (1<<bit) = mask ปิดบิตเป้าหมาย · | (wm << bit) = ฝังลายน้ำ

เลือก bit

bit ต่ำ (0) ซ่อนแนบเนียนแต่เปราะ · bit สูง (7) เห็นชัด — โจทย์กำหนด embed_bit ให้

เทคนิคสอบขั้นตอน: & mask (ล้างบิตเดิม) → | wm<<bit (ฝัง) — สลับไม่ได้

WEEK 4 · Ex2

MSE / PSNR — วัดคุณภาพภาพ

เปรียบเทียบภาพต้นฉบับกับภาพที่บีบอัด/quantize แล้ว

def mse(original, quantized):
    e = original.astype(float) - quantized.astype(float)
    return np.mean(e ** 2)

def psnr(original, quantized):
    err = mse(original, quantized)
    if err == 0: return float("inf")
    return 10 * np.log10((255 ** 2) / err)

# ลูปทดสอบทุกระดับ
for level in [256, 128, 64, 32, 16, 8, 4, 2]:
    q = quantize(gray, level, "uniform", 2)
    print(level, mse(gray, q), psnr(gray, q))

อ่านผล

MSE ยิ่งน้อยดี · PSNR ยิ่งมากดี (dB) · PSNR = 10·log10(255² / MSE)

แนวโน้ม

level เทาน้อยลง → MSE เพิ่ม / PSNR ลด (ภาพแย่ลง) — กราฟ inverted x-axis

เทคนิคสอบPSNR = ∞ เมื่อภาพเหมือนเป๊ะ (MSE = 0) — กันหารด้วย 0

WEEK 4 · Ex3

Intensity Transforms — Negative · Gamma · Log

ปรับค่าความเข้มพิกเซลด้วยสมการ — normalize เป็น 0–1 ก่อนเสมอ

# Negative (ฟิล์ม)
J = 255 - I                     # หรือ cv2.bitwise_not(I)

# Gamma / Power-Law  s = β·r^γ
def power_law(img, gamma=1, beta=1):
    r = img / 255.0
    s = np.clip(beta * (r ** gamma), 0, 1)
    return np.uint8(s * 255)
# γ<1 สว่าง · γ>1 มืด · γ=0.4 ใช้แก้ภาพมืด

# Log  s = c·log(1+r)
def log_transform(img, domain=255):
    f = img.astype(np.float32) / 255.0 * domain
    c = 1 / np.log(1 + domain)
    return c * np.log(1 + f)

# Contrast stretch แบบ sigmoid
def contrast_stretch(img, center=0.5, exponent=4):
    img = img.astype(np.float32) / 255.0
    eps = np.finfo(np.float32).eps
    return np.clip(1 / (1 + (center / (img + eps)) ** exponent), 0, 1)

สูตรต้องจำ

Negative: 255−r · Gamma: s = β·r^γ · Log: c·log(1+r), c = 1/log(1+domain)

ทำไม +1

log(0) หาค่าไม่ได้ → log(1+r) · log ขยายเงามืด (dynamic range กว้าง)

เทคนิคสอบnormalize 0–255 → 0–1 ก่อน แล้วคูณกลับ + clip(0,1) — กันค่าล้น

WEEK 4 · Ex4

Fourier Transform — FFT Workflow

fft2 → fftshift → (กรอง) → ifftshift → ifft2 — ลำดับนี้ห้ามสลับ

img = cv2.imread("cameraman.tif", 0)
img = img.astype(np.float32) / 255.0     # 0–1

F = np.fft.fft2(img)
F_shift = np.fft.fftshift(F)             # ย้าย DC มาอยู่กลาง
magnitude = np.log1p(np.abs(F_shift))    # spectrum
phase = np.angle(F_shift)                # phase

img_back = np.abs(np.fft.ifft2(F))       # inverse → ภาพเดิม
diff = np.abs(img - img_back)

อ่านสเปกตรัม

กลาง = DC/ความถี่ต่ำ · ขอบ = ความถี่สูง · magnitude = ความแรง, phase = ตำแหน่ง

กับดัก

ifft2 คืนค่า complex → ต้อง np.abs() · log1p = log(1+x) กัน log(0)

เทคนิคสอบfreq profile: ดึงแถวกลาง magnitude[y, :] พล็อตดูความถี่

WEEK 4 · Ex6

Walsh / Hadamard + Compression

สร้าง basis matrix ±1 แล้ว transform: F = (1/N)·H·img·H — ภาพต้องเป็น 2^n ก่อน

N = len(a)                # ต้อง resize เป็น 2^n (เช่น 256)
n = int(np.log2(N))
f = np.ones((N, N))
for x in range(N):
    for u in range(N):
        p = format(x, "0{}b".format(n))
        q = format(u, "0{}b".format(n))
        value = 1
        for i in range(n):
            # Walsh: q[i] · Hadamard: q[n-1-i]
            value *= (-1) ** (int(p[n-1-i]) * int(q[i]))
        f[x, u] = value

F = (1 / N) * f @ a.astype(float) @ f   # transform
G = np.fft.fftshift(F)
H = np.log(1 + np.abs(G))

# Compression: ตัดค่าสเปกตรัมเล็กทิ้ง (percentile)
th = np.percentile(np.abs(F), 90)
F_comp = F.copy()
F_comp[np.abs(F_comp) < th] = 0
rec = np.clip((1 / N) * f @ F_comp @ f, 0, 255).astype(np.uint8)

Walsh vs Hadamard

Walsh = sequency order (q[i]) · Hadamard = natural order (q[n-1-i]) — ต่างที่ index เดียว

Compression

ตัดค่าที่ < percentile ทิ้ง → inverse ใหม่ → เทียบ CR/MSE/PSNR

เทคนิคสอบmatrix @ ภาพ @ matrix (2 ฝั่ง) · หาร 1/N · clip 0–255 + uint8

WEEK 5 · Lab5_1

Periodic Noise + Notch Filter

noise แบบ sine → กลายเป็นจุดสว่างในสเปกตรัม → ปิดด้วย mask วงกลม

rows, cols = img.shape
x = np.arange(cols); y = np.arange(rows)
X, Y = np.meshgrid(x, y)
fx = fy = 80
noise = (A * np.sin(2*np.pi*fx*X/cols) +
         A * np.sin(2*np.pi*fy*Y/rows))
noisy = np.clip(img.astype(float) + noise, 0, 255).astype(np.uint8)

F_shift = np.fft.fftshift(np.fft.fft2(noisy))
mag = np.log1p(np.abs(F_shift))

# Notch mask — ปิด peak (ยกเว้น DC กลาง)
mask = np.ones((rows, cols), np.float32)
th = 0.8 * np.max(mag)
for y_, x_ in zip(*np.where(mag > th)):
    if abs(x_ - cols//2) > 5 or abs(y_ - rows//2) > 5:
        cv2.circle(mask, (x_, y_), size, 0, -1)

F_filtered = F_shift * mask
result = np.abs(np.fft.ifft2(np.fft.ifftshift(F_filtered)))
result = np.clip(result, 0, 255).astype(np.uint8)

หลักการ

sine ใน spatial = จุดสว่างคู่สมมาตร ในสเปกตรัม · ปิดจุดนั้น = ตัด noise

กับดัก

ห้ามปิด DC (กลางภาพ) — เป็นข้อมูลหลักของภาพ · ลำดับ fftshift → คูณ mask → ifftshift

เทคนิคสอบลำดับ: fft2 → fftshift → mask → ifftshift → ifft2 → abs — จดให้ขึ้นใจ

WEEK 5 · Lab5_1

Wiener Filter + Morphology

ปรับตาม variance ท้องถิ่น (เก็บขอบ เบลอพื้นเรียบ) และปรับรูปร่างภาพ binary

def wiener_filter(img, k=7, noise_var=500):
    img = img.astype(np.float32)
    mean    = cv2.boxFilter(img, -1, (k, k), normalize=True)
    mean_sq = cv2.boxFilter(img*img, -1, (k, k), normalize=True)
    var = np.maximum(mean_sq - mean*mean, 0)
    out = mean + (np.maximum(var - noise_var, 0) /
                  np.maximum(var, noise_var)) * (img - mean)
    return np.clip(out, 0, 255).astype(np.uint8)

# Morphology (ภาพ binary)
se = np.ones((5, 5), np.uint8)
erosion  = cv2.erode(img, se, iterations=1)    # หด
dilation = cv2.dilate(img, se, iterations=2)   # ขยาย
opening  = cv2.morphologyEx(img, cv2.MORPH_OPEN, se)   # E→D
closing  = cv2.morphologyEx(img, cv2.MORPH_CLOSE, se)  # D→E

Wiener logic

variance สูง = ขอบ → เก็บ · variance ต่ำ = เรียบ → เบลอ — ดีกว่า blur ตรง ๆ

จำให้ถูก

Opening = erode→dilate (ลบจุดเล็ก) · Closing = dilate→erode (ปิดรู) — อย่าสลับ

เทคนิคสอบerode = ทุกช่องต้องขาว (AND) · dilate = มีขาวจุดเดียวพอ (OR)
