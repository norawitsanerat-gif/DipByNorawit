9 · Lab3 — Contrast Stretch + Equalize (lab3_2)

# Piecewise linear stretch (แบ่ง 3 ช่วง) def pixelVal(pix, r1, s1, r2, s2): if 0 <= pix <= r1: return (s1 / r1) * pix elif r1 < pix <= r2: return ((s2 - s1) / (r2 - r1)) * (pix - r1) + s1 else: return ((255 - s2) / (255 - r2)) * (pix - r2) + s2 def contrast_stretch(img, r1, s1, r2, s2): return np.vectorize(lambda p: pixelVal(p, r1, s1, r2, s2))(img).astype(np.uint8) result = contrast_stretch(gray, 70, 20, 180, 230) equalized = cv2.equalizeHist(gray) # histogram equalization
  • np.vectorize = เอา function ไปรันทุกพิกเซล · สุดท้าย .astype(np.uint8) เสมอ
  • equalizeHist ใช้ได้เฉพาะ grayscale (1 channel)

10 · Week4 — Quantization + Downsampling (Ex1–Ex2)

# Uniform quantization → L ระดับ def quantize(img, levels, option=1): step = 256 // levels if option == 0: # Min q = (img // step) * step elif option == 1: # Mid (ค่ากลางช่วง) q = (img // step) * step + step // 2 elif option == 2: # Max q = np.minimum((img // step) * step + step - 1, 255) return q.astype(np.uint8) # 16 ระดับ: q = (img // 16) * 16 + 7 # Downsampling — เลือกทุก ๆ n พิกเซล z = cv2.resize(cv2.imread("cameraman.tif", 0), (1024, 1024)) a = z[::2, ::2]; b = z[::4, ::4] # ¼ พิกเซล, 1/16 พิกเซล e = z[::16, ::16]; d = z[::32, ::32]
  • Mid = ตัวแทนช่วง (step//2) — Ex1 ใช้ (img//16)*16+7 กับดัก: ไม่ใช่ *16 ตรง ๆ
  • levels ลง 2 ระดับ (256→128→64…) ภาพหยาบขึ้นเรื่อย ๆ

from PIL import Image, ImageDraw, ImageFont # สร้าง layer ข้อความขาว-ดำ def create_watermark(text, shape, position=(50,50), alpha=255): h, w = shape wm = Image.new("L", (w, h), 0) draw = ImageDraw.Draw(wm) try: font = ImageFont.truetype("arial.ttf", 30) except: font = ImageFont.load_default() draw.text(position, text, fill=alpha, font=font) return np.array(wm) wm = create_watermark("Norawit S.", img.shape, (50, 100), 255) wm_bin = (wm > 128).astype(np.uint8) # ฝัง: ปิดบิตเป้าหมาย แล้ว OR ใส่ watermark embed_bit = 7 # เลือก bit 0–7 mask = np.uint8(255 ^ (1 << embed_bit)) # เปิดทุกบิต ยกเว้นบิตเป้าหมาย result = img.copy() result = (result & mask) | (wm_bin << embed_bit) # ตรวจ/ดึงกลับ bit_plane = ((result >> embed_bit) & 1) * 255 reconstructed = result & (1 << embed_bit)

13 · Week4 — วัดคุณภาพ MSE / PSNR (Ex2)

def mse(original, quantized): e = original.astype(float) - quantized.astype(float) return np.mean(e ** 2) def psnr(original, quantized): err = mse(original, quantized) if err == 0: return float("inf") return 10 * np.log10((255 ** 2) / err) # ลูปหาคุณภาพทุก level for level in [256, 128, 64, 32, 16, 8, 4, 2]: q = quantize(gray, level, "uniform", 2) print(level, mse(gray, q), psnr(gray, q))
  • MSE ยิ่งน้อยดี · PSNR ยิ่งมากดี (dB) · PSNR = 10·log10(255²/MSE)
  • ระดับเทาน้อยลง → MSE เพิ่ม, PSNR ลด

14 · Week4 — Intensity Transforms (Ex3)

# Negative (ฟิล์ม) J = 255 - I # หรือ cv2.bitwise_not(I) # Gamma / Power-Law s = β·r^γ def power_law(img, gamma=1, beta=1): r = img / 255.0 s = np.clip(beta * (r ** gamma), 0, 1) return np.uint8(s * 255) # γ<1 สว่าง · γ>1 มืด · γ=0.4 ใช้แก้ภาพมืด # Log s = c·log(1+r) def log_transform(img, domain=255): f = img.astype(np.float32) / 255.0 * domain c = 1 / np.log(1 + domain) return c * np.log(1 + f) # Contrast stretch แบบ sigmoid def contrast_stretch(img, center=0.5, exponent=4): img = img.astype(np.float32) / 255.0 eps = np.finfo(np.float32).eps return np.clip(1 / (1 + (center / (img + eps)) ** exponent), 0, 1)
  • normalize 0–255 → 0–1 ก่อนเสมอ แล้วคูณกลับ · np.clip(…, 0, 1) กันเกิน
  • log ขยายเงามืด (dynamic range กว้าง) · ทำไม +1 → log(0) หาค่าไม่ได้

15 · Week4 — Fourier / FFT (Ex4)

img = cv2.imread("cameraman.tif", 0) img = img.astype(np.float32) / 255.0 # 0–1 F = np.fft.fft2(img) F_shift = np.fft.fftshift(F) # ย้าย DC มาอยู่กลาง magnitude = np.log1p(np.abs(F_shift)) # spectrum (log ดึงรายละเอียด) phase = np.angle(F_shift) # phase img_back = np.abs(np.fft.ifft2(F)) # inverse → ภาพเดิม diff = np.abs(img - img_back)

  • ifft2 แล้วต้อง np.abs() (ผลเป็น complex)

17 · Week4 — Walsh / Hadamard (Ex6)

N = len(a) # ต้อง resize เป็น 2^n ก่อน (เช่น 256) n = int(np.log2(N)) # หรือ 1 + floor(log2(N)) f = np.ones((N, N)) for x in range(N): for u in range(N): p = format(x, "0{}b".format(n)); q = format(u, "0{}b".format(n)) value = 1 for i in range(n): # Walsh: int(q[i]) · Hadamard: int(q[n-1-i]) value *= (-1) ** (int(p[n-1-i]) * int(q[i])) f[x, u] = value F = (1 / N) * f @ a.astype(float) @ f # transform G = np.fft.fftshift(F) H = np.log(1 + np.abs(G)) # Compression: ตัดค่าสเปกตรัมที่เล็ก (เปอร์เซ็นไทล์) ทิ้ง th = np.percentile(np.abs(F), 90) F_comp = F.copy(); F_comp[np.abs(F_comp) < th] = 0 rec = np.clip((1 / N) * f @ F_comp @ f, 0, 255).astype(np.uint8)
  • Walsh = sequency order (q[i]) · Hadamard = natural order (q[n-1-i]) — ต่างที่ index
  • matrix @ ภาพ @ matrix (คูณ 2 ฝั่ง) · หาร 1/N · สุดท้าย clip 0–255 + uint8
  • basis เป็น ±1 (square wave) · ใช้คำนวณ compression ratio = ขนาดเดิม/ขนาดใหม่

22 · week5 — Wiener Filter + Morphology (Lab5_1)

# Wiener — ปรับตาม variance ท้องถิ่น def wiener_filter(img, k=7, noise_var=500): img = img.astype(np.float32) mean = cv2.boxFilter(img, -1, (k, k), normalize=True) mean_sq = cv2.boxFilter(img*img, -1, (k, k), normalize=True) var = np.maximum(mean_sq - mean*mean, 0) out = mean + (np.maximum(var - noise_var, 0) / np.maximum(var, noise_var)) * (img - mean) return np.clip(out, 0, 255).astype(np.uint8) # Morphology (ภาพ binary) se = np.ones((5, 5), np.uint8) erosion = cv2.erode(img, se, iterations=1) # หด dilation = cv2.dilate(img, se, iterations=2) # ขยาย opening = cv2.morphologyEx(img, cv2.MORPH_OPEN, se) # erode→dilate ลบจุดเล็ก closing = cv2.morphologyEx(img, cv2.MORPH_CLOSE, se) # dilate→erode ปิดรู
  • Wiener: variance สูง = ขอบ → เก็บรายละเอียด · variance ต่ำ = พื้นเรียบ → เบลอ
  • Opening = E→D (ลบจุดขาวเล็ก) · Closing = D→E (ปิดรูดำ)
